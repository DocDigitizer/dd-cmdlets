DocDigitizer PowerShell Module

Scripting Examples

Installation | Commands Reference | Scripting Examples

Quick Start

# Import the module
Import-Module .\dd-cmdlets\DocDigitizer.psd1

# Test connection
Test-DocDigitizerConnection

# Process a document (returns JSON)
Send-DocDigitizerDocument -FilePath "invoice.pdf"

# Process and auto-save to invoice_extraction.json
Send-DocDigitizerDocument -FilePath "invoice.pdf" -SaveExtraction

Basic Document Processing

Process a Single Document

# Simple processing - returns JSON
Send-DocDigitizerDocument -FilePath "invoice.pdf"

# Process with specific IDs for tracking
$docId = [guid]::NewGuid()
$contextId = [guid]::NewGuid()
Send-DocDigitizerDocument -FilePath "invoice.pdf" -DocumentId $docId -ContextId $contextId

Save Results to File

# Auto-save to {filename}_extraction.json in same directory
Send-DocDigitizerDocument -FilePath "invoice.pdf" -SaveExtraction
# Creates: invoice_extraction.json

# Save to custom path
Send-DocDigitizerDocument -FilePath "invoice.pdf" -OutputPath "C:\results\my_result.json"

Pipeline Selection

# Use a specific pipeline
Send-DocDigitizerDocument -FilePath "invoice.pdf" -Pipeline "MainPipelineWithOCR"

# Available pipelines (check your server):
# - MainPipelineWithOCR      (OCR + extraction)
# - MainPipelineWithFile     (extraction only, for searchable PDFs)
# - SingleDocPipelineWithOCR
# - SingleDocPipelineWithFile

Response Verbosity

# Minimal response (default) - just essential data
Send-DocDigitizerDocument -FilePath "invoice.pdf"

# Medium response - includes metadata
Send-DocDigitizerDocument -FilePath "invoice.pdf" -LogLevel Medium

# Full response - complete execution details
Send-DocDigitizerDocument -FilePath "invoice.pdf" -LogLevel Full

Batch Processing

Process All PDFs in a Folder

# Process all PDFs with the same context ID (groups them together)
$contextId = [guid]::NewGuid()
Get-ChildItem *.pdf | Send-DocDigitizerDocument -ContextId $contextId

Process and Save All Results

# Process all PDFs and save each result alongside the original
Get-ChildItem *.pdf | Send-DocDigitizerDocument -SaveExtraction

Collect Results for Further Processing

# Collect all results into a variable
$results = Get-ChildItem *.pdf | Send-DocDigitizerDocument -ContextId $contextId

# Export summary to CSV
$results | ForEach-Object { $_ | ConvertFrom-Json } |
    Select-Object traceId, state, pageCount, filePath |
    Export-Csv -Path "processing_summary.csv" -NoTypeInformation

Working with JSON Results

# Get JSON result
$json = Send-DocDigitizerDocument -FilePath "invoice.pdf"

# Parse JSON to PowerShell object
$result = $json | ConvertFrom-Json

# Access specific fields
$result.traceId
$result.output.extractions
$result.timers

# Pretty-print the output
$result.output | ConvertTo-Json -Depth 10

Health Check and Connectivity

# Detailed connection test
Test-DocDigitizerConnection

# Simple boolean check for scripts
if (Test-DocDigitizerConnection -Quiet) {
    Write-Host "API is available"
    Send-DocDigitizerDocument -FilePath "invoice.pdf"
} else {
    Write-Host "API is not available"
}

Configuration Management

View Current Configuration

Get-DocDigitizerConfig

Set Configuration for Current Session

# Set pipeline and log level
Set-DocDigitizerConfig -Pipeline "MainPipelineWithOCR" -LogLevel Full

# Set custom timeout for large documents
Set-DocDigitizerConfig -Timeout 600

Persist Configuration

# Save to PowerShell profile (survives session restarts)
Set-DocDigitizerConfig -Pipeline "MainPipelineWithOCR" -Persist

Use Different Environments

# Switch to local development server
Set-DocDigitizerConfig -BaseUrl "http://localhost:5000"

# Switch back to production
Set-DocDigitizerConfig -BaseUrl "https://apix.docdigitizer.com/sync"

# One-off call to different server (without changing config)
Send-DocDigitizerDocument -FilePath "doc.pdf" -BaseUrl "http://localhost:5000"

Error Handling

Silent Error Handling

# Continue on error, check result
$result = Send-DocDigitizerDocument -FilePath "document.pdf" -ErrorAction SilentlyContinue
if (-not $result) {
    Write-Host "Processing failed"
}

Try/Catch Error Handling

try {
    $result = Send-DocDigitizerDocument -FilePath "document.pdf" -ErrorAction Stop
    Write-Host "Success! TraceId: $(($result | ConvertFrom-Json).traceId)"
} catch {
    Write-Host "Error: $($_.Exception.Message)"
}

Batch Processing with Error Handling

$files = Get-ChildItem *.pdf
$successful = @()
$failed = @()

foreach ($file in $files) {
    try {
        $result = Send-DocDigitizerDocument -FilePath $file.FullName -ErrorAction Stop
        $successful += @{ File = $file.Name; Result = $result }
    } catch {
        $failed += @{ File = $file.Name; Error = $_.Exception.Message }
    }
}

Write-Host "Processed: $($successful.Count) successful, $($failed.Count) failed"

Advanced Scripting

Process Documents from a List

# Read file paths from a text file
$files = Get-Content "documents_to_process.txt"
$contextId = [guid]::NewGuid()

$files | ForEach-Object {
    if (Test-Path $_) {
        Send-DocDigitizerDocument -FilePath $_ -ContextId $contextId -SaveExtraction
    } else {
        Write-Warning "File not found: $_"
    }
}

Scheduled Processing Script

# watch_folder.ps1 - Process new PDFs in a folder
param(
    [string]$WatchFolder = "C:\Incoming",
    [string]$ProcessedFolder = "C:\Processed"
)

Import-Module DocDigitizer

$pdfs = Get-ChildItem $WatchFolder -Filter "*.pdf"

foreach ($pdf in $pdfs) {
    Write-Host "Processing: $($pdf.Name)"

    try {
        Send-DocDigitizerDocument -FilePath $pdf.FullName -SaveExtraction -ErrorAction Stop
        Move-Item $pdf.FullName $ProcessedFolder
        Write-Host "  Done - moved to processed folder"
    } catch {
        Write-Warning "  Failed: $($_.Exception.Message)"
    }
}

Generate Processing Report

# Generate an HTML report of processed documents
$results = Get-ChildItem *.pdf | Send-DocDigitizerDocument -ContextId ([guid]::NewGuid())

$report = $results | ForEach-Object {
    $r = $_ | ConvertFrom-Json
    [PSCustomObject]@{
        File = Split-Path $r.filePath -Leaf
        TraceId = $r.traceId
        State = $r.state
        Pages = $r.pageCount
        Duration = "$([math]::Round($r.timers.'DocWorker_Total' / 1000, 2))s"
    }
}

$report | ConvertTo-Html -Title "Processing Report" | Out-File "report.html"
Write-Host "Report saved to report.html"
